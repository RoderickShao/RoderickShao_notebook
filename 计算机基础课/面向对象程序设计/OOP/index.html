    xchen_algorithm.h

    #pragma once
#include <iostream>

namespace xchen {

template<typename T>
int min_element(T arr[], int begin, int end)
{
  int min_idx = begin;
  for (int i = begin+1; i < end; ++i)
  {
    if (arr[i] < arr[min_idx])
      min_idx = i;
  }
  return min_idx;
}

template<typename T, typename Compare>
int min_element(T arr[], int begin, int end, Compare comp)
{
  int min_idx = begin;
  for (int i = begin+1; i < end; ++i)
  {
    if (comp(arr[i], arr[min_idx]))
      min_idx = i;
  }
  return min_idx;
}

template<typename T>
void swap(T& a, T& b)
{
  T tmp = a;
  a = b;
  b = tmp;
}

template<typename T>
void selection_sort(T arr[], int n)
{
  for (int i = 0; i < n-1; ++i)
  {
    int min_idx = min_element(arr, i, n);
    if (min_idx != i)
      swap(arr[min_idx], arr[i]);
  }
}

template<typename T, typename Compare>
void selection_sort(T arr[], int n, Compare comp)
{
  for (int i = 0; i < n-1; ++i)
  {
    int min_idx = min_element(arr, i, n, comp);
    if (min_idx != i)
      swap(arr[min_idx], arr[i]);
  }
}

template<typename T>
void print_array(T arr[], int n)
{
  for (int i = 0; i < n; ++i)
    std::cout << arr[i] << ' ';
  std::cout << '\n';
}

template<typename T>
void print_array(T* arr[], int n)
{
  for (int i = 0; i < n; ++i)
    std::cout << *arr[i] << ' ';
  std::cout << '\n';
}

} // end of namespace xchen

    #include <cmath>
#include <iostream>
#include <iomanip>
#include <string>
#include "xchen_algorithm.h"

const double PI = 3.14;
class Shape {
protected:
  double area, perimeter;

public:
  virtual ~Shape() {}
  virtual void calc_area() = 0;
  virtual void calc_perimeter() = 0;
  virtual std::string name() const = 0;

  double get_area() const { return area; }
  double get_perimeter() const { return perimeter; }

  friend std::ostream& operator<<(std::ostream&, const Shape&);
};

class Rectangle : public Shape {
private:
  double w, h;

public:
  Rectangle(double w, double h) : w(w), h(h) {}
  void calc_area() override {
    area = w * h;
  }
  void calc_perimeter() override {
    perimeter = 2 * (w + h);
  }
  std::string name() const override {
    return "Rectangle";
  }
};

class Circle : public Shape {
private:
  double r;

public:
  Circle(double r) : r(r) {}
  void calc_area() override {
    area = PI * r * r;
  }
  void calc_perimeter() override {
    perimeter = 2 * PI * r;
  }
  std::string name() const override {
    return "Circle";
  }
};

class Triangle : public Shape {
private:
  double a, b, c;

public:
  Triangle(double a, double b, double c) : a(a), b(b), c(c) {}
  void calc_area() override {
    double s = (a + b + c) / 2;
    area = sqrt(s * (s - a) * (s - b) * (s - c));
  }
  void calc_perimeter() override {
    perimeter = a + b + c;
  }
  std::string name() const override {
    return "Triangle";
  }
};

std::ostream& operator<<(std::ostream& out, const Shape& s)
{
  return out << "(" << s.name() << ":" << s.area << "," << s.perimeter << ")";
}

bool less_shape_area(Shape* s1, Shape* s2)
{
  return s1->get_area() < s2->get_area();
}

bool less_shape_perimeter(Shape* s1, Shape* s2)
{
  return s1->get_perimeter() < s2->get_perimeter();
}

bool greater_shape_area(Shape* s1, Shape* s2)
{
  return s1->get_area() > s2->get_area();
}

void print_sorting_head(const std::string& elem_type)
{
  std::cout << "\n===== sorting [" << elem_type << "] array =====\n";
}

void print_sorting_option(const std::string& option)
{
  std::cout << std::setw(20) << std::left << ("by " + option + ":");
}

int main()
{
    print_sorting_head("Shape*");
    Shape* arr[] = {new Rectangle(2,3), new Rectangle(5,5), new Circle(3), new Triangle(2,5,4)};
    int n = sizeof(arr) / sizeof(arr[0]);
    for (Shape* s : arr)
    {
      s->calc_area();
      s->calc_perimeter();
    }
    xchen::selection_sort(arr, n, less_shape_area);
    print_sorting_option("area<");
    xchen::print_array(arr, n);
    xchen::selection_sort(arr, n, less_shape_perimeter);
    print_sorting_option("perimeter<");
    xchen::print_array(arr, n);
    xchen::selection_sort(arr, n, greater_shape_area);
    print_sorting_option("area>");
    xchen::print_array(arr, n);
    xchen::selection_sort(arr, n, [](Shape* s1, Shape* s2) { return s1->get_perimeter() > s2->get_perimeter(); });
    print_sorting_option("perimeter>");
    xchen::print_array(arr, n);

    for (Shape* s : arr)
      delete s;

  return 0;
}

    3.1 Container¶

    Sequence Containers (顺序容器)¶

    这些容器按照元素插入的顺序存储元素，并允许根据位置访问元素。

    (1) Array (静态数组):

    • 是一个固定大小的容器，一旦创建就不能改变其大小。
    • 提供直接访问元素的能力（通过索引），效率非常高。
    • 适用于已知大小的数据集合，且在程序运行期间不会变化的情况。

    (2) Vector (动态数组):

    • 类似于数组，但可以动态调整大小。
    • 在尾部插入和删除元素非常高效，但在中间或头部进行插入和删除操作效率较低，因为需要移动元素。
    • 适用于大多数情况下需要快速随机访问元素的应用场景。

    (3) Deque (双端队列):

    • 支持从两端高效地插入和删除元素。
    • 相比于vector，在头部和尾部的操作更加高效，但中间部分的操作效率较低。
    • 适用于需要频繁在两端添加或移除元素的场景。

    (4) Forward List (单向链表)

    • 单向链表，每个节点包含一个指向下一个节点的指针。
    • 只能从前向后遍历。
    • 插入和删除操作高效，但不支持随机访问。

    (5) List (双向链表)

    • 双向链表，每个节点包含指向前一个节点和后一个节点的指针。
    • 支持双向遍历。
    • 插入和删除操作高效，但不支持随机访问。

    Associative Containers (关联容器)¶

    这些容器存储元素时会自动根据某个键值排序，并且不允许有重复的键（除非是multi-set或multi-map）。

    (1) Set, Multi-set:

    • set存储唯一的键值，不允许重复。
    • multi-set允许存储重复的键值。
    • 元素自动按升序排列（默认情况），支持高效的查找、插入和删除操作。

    (2) Map, Multi-map:

    • map是一种键值对容器，每个键对应一个值，键唯一。
    • multi-map允许相同的键对应多个值。
    • 元素根据键自动排序，提供了基于键的高效查找、插入和删除操作。

    Unordered Associative Containers (无序关联容器)¶

    这些容器使用哈希表来存储数据，因此不保证元素的顺序。它们提供平均情况下O(1)的查找、插入和删除性能。

    (1) Unordered_set:

    • 存储唯一的键值，类似于set，但是元素没有特定的顺序。
    • 使用哈希函数来组织数据，适合需要快速查找的场景。

    (2) Unordered_map:

    • 键值对形式存储，类似于map，但不保持任何特定的顺序。
    • 同样利用哈希函数来提高查找速度，适用于需要快速访问元素的场景。

    Container Adaptors (容器适配器)¶

    这些不是独立的容器类型，而是基于其他容器实现的接口，用于提供特定的功能。

    Stack (栈)

    • 提供后进先出（LIFO）的数据结构。
    • 默认基于deque实现，但也可以基于vector或list实现。
    • 主要操作包括push（入栈）、pop（出栈）和top（查看栈顶元素）。

    #include<iostream>
#include<vector>
using namespace std;

int main(){
    //vector 容器的初始化
    vector<int> evens{2,4,6,8};
    //加入数据，push_back与insert
    evens.push_back(20);
    evens.push_back(22);
    //指定位置（末尾）之前插入5个10，还有很多用法
    evens.insert(evens.begin()+4,5,10);

    //输出vector中的内容
    for(int i = 0;i<evens.size();++i)
        cout << evens[i] << ' ';
    cout << endl;

    //利用迭代器输出， it指向的就是容器中的数据
    for(vector<int>::iterator it = evens.begin();it<evens.end();it++)//if list,we can't use < iterator,we can use != iterator
        cout << *it << ' ';
    cout << endl;

    //遍历元素
    for(int e:evens)//迭代器的作用
        cout<< e << ' ';
}

bool is_balanced(const string &s){
    stack<char> st;
    for(char c:s){
        if(c=='('||c=='{'c=='['){
            st.push(c);
        }
        else if(c==')'||c=='}'||c==']'){
            if (st.empty())
                return false;
            char top= st.top();
            st.pop();
            if((c==')'&&top !='(')||(c==']'&&top!='[')||(c=='}'&&top!='{')
                return false;
        }
    }
}
